plotSectors <- function(sectors = c("Total", "Primary industries", "Goods producing industries", "Services industries"), startYear, endYear, df = read.csv("gdp-nz.csv", stringsAsFactors = TRUE), col = c("#2297E6", "#DF536B"))
  {
    #Checking validity of sectors and colours
    stopifnot(length(sectors) > 0)
    stopifnot(is.character(sectors) | is.numeric(sectors))
    stopifnot(!anyNA(sectors))
    stopifnot(is.character(col), length(col) == 2, !anyNA(col))

    #Setting up plot layout
    ncol <- min(3, length(sectors))
    nrow <- ceiling(length(sectors) / ncol)
    par(mfrow = c(nrow, ncol))

    #Plotting GDP growth rates for each sector
    for (sector in sectors)
      {
        plotGrowths(sector, startYear, endYear, df, col = col)
      }
  }
